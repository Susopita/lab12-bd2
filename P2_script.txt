-- P2.3: Distribución de contrataciones por año

SELECT
    EXTRACT(YEAR FROM hire_date)::INT AS anio,
    COUNT(*) AS total_empleados
FROM employees
GROUP BY EXTRACT(YEAR FROM hire_date)
ORDER BY anio;

-- P2.3: Rango temporal del dataset

SELECT
    MIN(hire_date) AS primera_contratacion,
    MAX(hire_date) AS ultima_contratacion,
    COUNT(*) AS total_empleados,
    COUNT(*) FILTER (WHERE hire_date IS NULL) AS fechas_nulas
FROM employees;

-- P2.4 - EXPERIMENTO C
-- PARTICIONAMIENTO MENSUAL
CREATE TABLE employees2c (
    LIKE employees
)
PARTITION BY RANGE (hire_date);

-- Crear una particion por mes

DO $$
DECLARE
    fecha_inicio DATE := DATE '1985-01-01';
    fecha_final DATE := DATE '2000-02-01';
    fecha_actual DATE;
    fecha_siguiente DATE;
    nombre_particion TEXT;
BEGIN
    fecha_actual := fecha_inicio;

    WHILE fecha_actual < fecha_final LOOP

        fecha_siguiente :=
            (fecha_actual + INTERVAL '1 month')::DATE;

        nombre_particion :=
            'employees2c_' ||
            TO_CHAR(fecha_actual, 'YYYY_MM');

        EXECUTE format(
            'CREATE TABLE %I PARTITION OF employees2c
             FOR VALUES FROM (%L) TO (%L)',
            nombre_particion,
            fecha_actual,
            fecha_siguiente
        );

        fecha_actual := fecha_siguiente;

    END LOOP;
END $$;


INSERT INTO employees2c
SELECT * FROM employees;
